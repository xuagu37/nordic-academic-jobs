<script>
document.addEventListener("DOMContentLoaded", function () {
  const typeSelect = document.getElementById('filterType');
  const textInput = document.getElementById('jobFilter');
  const jobBlocks = document.querySelectorAll('.job');

  // Save original HTML so we can restore before highlighting
  jobBlocks.forEach(job => {
    const titleEl = job.querySelector('h3');
    if (titleEl) {
        titleEl.dataset.originalText = titleEl.textContent;
    }
  });

  function escapeRegExp(string) {
    return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  }

  function highlightText(job, query) {
    const titleEl = job.querySelector('h3');
    if (!titleEl) return;
    
    const original = titleEl.dataset.originalText || titleEl.textContent;
    titleEl.dataset.originalText = original;
    
    if (!query) {
        titleEl.textContent = original;
        return;
    }
    
    const safeQuery = escapeRegExp(query);
    const regex = new RegExp(`(${safeQuery})`, 'gi');
    
    titleEl.innerHTML = original.replace(regex, '<mark>$1</mark>');
  }

  function updateDisplay() {
    const selectedType = typeSelect.value.toLowerCase();
    const query = textInput.value.trim().toLowerCase();

    jobBlocks.forEach(job => {
        
      const jobType = (job.dataset.type || "").toLowerCase();
      const titleEl = job.querySelector('h3');
      const titleText = titleEl ? titleEl.textContent.toLowerCase() : "";

      const matchesType = !selectedType || jobType === selectedType;
      const matchesQuery = !query || titleText.includes(query);

      if (matchesType && matchesQuery) {
        job.style.display = '';
        highlightText(job, query);
      } else {
        job.style.display = 'none';
      }
    });
  }

  typeSelect.addEventListener('change', updateDisplay);
  textInput.addEventListener('input', updateDisplay);
  updateDisplay();
});
</script>
